Fem servir el \textbf{canvi de bases} per conveniència, quan el sistema es fa més fàcil. \textbf{VECTORS CIRCULARS}\\
\\
$\vec{e_r} = \cos\varphi \vec{i} + \sin\varphi \vec{j} $\\
$\vec{e_{\varphi}} = -\sin\varphi \vec{i} + \cos \varphi \vec{j} $\\
$\dv{}{t} \vec{e_r} = \dot{\varphi} \vec{e_{\varphi}} $\\
$\dv{}{t} \vec{e_{\varphi}} = -\dot{\varphi} \vec{e_r} $\\
\\
\textbf{EQUACIÓ DE VECTOR POSICIÓ}\\ 
\\
$\vec{r}(t) = R \vec{e_r}$\\
\\
\textbf{VELOCITAT}\\
\\
$\vec{v} = R \dot{\varphi}\vec{e_{\varphi}} \therefore v = \omega R$ \\
\\
\textbf{ACCELERACIÓ}\\ 
\\
$\vec{a} = R\ddot{\varphi}\vec{e_{\varphi}} - R\dot{\varphi}\dot{\varphi}\vec{e_r}$\\
$\vec{a_T} = R\alpha$(En direcció tangencial)\\ 
$\vec{a_N} = \frac{v^2}{R} = \omega^2 R$(En direcció cap a dins del cercle)\\

\end{tcolorbox}

\subsection{Moviments Generals}

\textbf{VECTORS DE FRENET}\\ 
\\
$\vec{e_1} = \frac{\vec{v}}{\left|\vec{v}\right|}$\\
$\vec{e_2} = \frac{\vec{\dot{\vec{e_1}}}}{\left|\vec{\dot{\vec{e_1}}}\right|}$ ($\leftrightarrow \vec{e_1} = (a,b) \therefore \vec{e_2} = (-b,a)$)\\
\\
\textbf{POSICIÓ}\\
\\
$\vec{r}(t) = x(t)\vec{i} + y(t)\vec{j} + z(t)\vec{k}$ \\
\\
\textbf{VELOCITAT}\\ 
\\
$\vec{v}(t) = v \vec{e_1}$\\
\\
\textbf{ACCELERACIÓ}\\
\\
$\vec{a}(t) = \dot{\vec{v}}(t) = \dot{v}\vec{e_1} + v\dot{\vec{e_1}} = \dot{v}\vec{e_1} + \frac{v^2}{R}$\\
\\
\textbf{RADI DE CORBATURA}\\
\\
$R = \frac{\left|\vec{v}\right|}{\left|\dot{\vec{e_1}}\right|} = k^{-1}$\\
\\
\textbf{DISTÀNCIA}\\
\\
$D = \int_{t_0}^{t} v(t) dt$\\
$D = \int_{x_0}^{x} \sqrt{1 + y'} dx$\\

\end{tcolorbox}
